Why CLAD?

🛠

Production-Grade Kernel

Built on OpenCASCADE (OCCT), the same geometry kernel used by FreeCAD, KiCad, and industrial CAD systems.

📊

Parametric by Default

Every dimension is a parameter. Change one value, regenerate the entire model. Perfect for design iteration.

🔍

Powerful Selectors

Select faces, edges, and features by direction, type, position, or Boolean combinations. No manual face picking.

📏

GD&T / Tolerancing

Full ASME Y14.5 support with export to STEP AP242 PMI. Add position, flatness, perpendicularity and more.

🔧

Real Thread Geometry

119 thread specifications (ISO Metric, UNC, UNF) with actual 3D helical geometry, not cosmetic.

📦

Assembly Support

Define assemblies with mate constraints, generate BOM, and export complete assemblies to STEP.

Code Examples

See how naturally designs translate to code with CLAD's declarative DSL.

;; Create a simple bracket in just a few lines
(clad.dsl:defpart bracket ((width 100) (height 50) (thickness 8))
  "Mounting bracket with bolt holes"
  (:body (clad.core:make-box width height thickness))

  ;; Add holes on the top face
  (:on-face :direction :+z :extreme :max
    (:circular-pattern :count 4 :radius 35
                       :center-x 50 :center-y 25
      (:cut (clad.core:make-cylinder 4 20)))))

;; Create and export
(clad.export:export-step (bracket) "bracket.step")
;; Parametric design - change one value, update everything
(clad.dsl:defpart flange ((pipe-dia 40) (flange-dia 80)
                          (thickness 12) (bolt-count 6))
  "Parametric pipe flange"
  (:body (clad.core:make-cylinder (/ flange-dia 2) thickness))

  ;; Central bore scales with pipe diameter
  (:cut (clad.core:make-cylinder (/ pipe-dia 2) (+ thickness 2)))

  ;; Bolt pattern automatically adjusts
  (:on-face :direction :+z :extreme :max
    (:circular-pattern
        :count bolt-count
        :radius (* flange-dia 0.4)
      (:cut (clad.core:make-cylinder 5 (+ thickness 2))))))

;; Create variations
(flange)                          ; Default: 40mm pipe, 6 bolts
(flange :pipe-dia 60 :bolt-count 8)  ; Larger version
;; Powerful face/edge selection without manual picking
(clad.dsl:defpart housing ((size 100))
  "Housing with features on selected faces"
  (:body (clad.core:make-box size size 50))

  ;; Select top face by direction
  (:on-face :direction :+z :extreme :max
    (:cut (clad.core:make-cylinder 20 60)))

  ;; Select all vertical faces
  (:on-faces :parallel :z
    (:add (clad.core:make-box 10 10 5)))

  ;; Combine selectors with AND/OR
  (:on-face :and :type :plane
                 :direction :+x
                 :extreme :max
    (:datum "A"))

  ;; Select by position
  (:on-edges :at-z 25 :tolerance 0.1
    (:fillet 3)))
;; GD&T with ASME Y14.5 compliance and STEP AP242 export
(clad.dsl:defpart precision-block ()
  "Precision machined block with full tolerancing"
  (:body (clad.core:make-box 100 50 30))

  ;; Primary datum on bottom face
  (:datum "A" :on-face :direction :-z :extreme :min)

  ;; Secondary datum on back face
  (:datum "B" :on-face :direction :-y :extreme :min)

  ;; Flatness on top surface
  (:flatness :on-face :direction :+z :extreme :max
             :tolerance 0.02)

  ;; Perpendicularity relative to datum A
  (:perpendicularity :on-face :direction :+y :extreme :max
                     :tolerance 0.05
                     :datum-ref "A")

  ;; True position for bolt hole pattern
  (:on-face :direction :+z :extreme :max
    (:circular-pattern :count 4 :radius 35
      (:cut (clad.core:make-cylinder 5 35))
      (:position :tolerance 0.1
                 :datum-refs ("A" "B")
                 :mmc t))))

;; Export with PMI data
(clad.export:export-step-ap242 (precision-block) "block.step")
;; 2D Sketch with constraints, then extrude or revolve
;; Create a sketch
(let* ((sketch (clad.sketch:make-sketch))
       ;; Define points
       (p1 (clad.sketch:make-point-2d 0 0 :fixed t))
       (p2 (clad.sketch:make-point-2d 80 0))
       (p3 (clad.sketch:make-point-2d 80 40))
       (p4 (clad.sketch:make-point-2d 0 40)))

  ;; Add lines forming rectangle
  (clad.sketch:add-entity sketch (clad.sketch:make-line-2d p1 p2))
  (clad.sketch:add-entity sketch (clad.sketch:make-line-2d p2 p3))
  (clad.sketch:add-entity sketch (clad.sketch:make-line-2d p3 p4))
  (clad.sketch:add-entity sketch (clad.sketch:make-line-2d p4 p1))

  ;; Extrude to 3D solid
  (clad.sketch:extrude-sketch sketch 20.0))

;; Revolution example - create a torus
(let* ((sketch (clad.sketch:make-sketch))
       (center (clad.sketch:make-point-2d 30 0))
       (circle (clad.sketch:make-circle-2d center 8.0)))
  (clad.sketch:add-entity sketch circle)
  ;; Revolve around Y axis
  (clad.sketch:revolve-sketch sketch))

Getting Started

1 Install Dependencies 
# Ubuntu/Debian
sudo apt install libocct-foundation-dev libocct-modeling-algorithms-dev \
                 libocct-data-exchange-dev sbcl

# macOS
brew install opencascade sbcl
2 Clone and Build 
git clone https://github.com/anthropics/clad.git
cd clad/c-wrapper && mkdir build && cd build
cmake .. && make
3 Load in REPL 
(push #p"/path/to/clad/" asdf:*central-registry*)
(asdf:load-system :clad)
4 Create Your First Part 
(clad.dsl:defpart my-first-part ()
  (:body (clad.core:make-box 50 30 20)))

(clad.export:export-step (my-first-part) "my-part.step")
